(function () { var KEY = "ys-theme"; var root = document.documentElement; var btn = document.getElementById("theme-toggle"); function current() { return root.getAttribute("data-theme") === "dark" ? "dark" : "light"; } function apply(theme) { root.setAttribute("data-theme", theme); if (btn) btn.setAttribute("aria-label", theme === "dark" ? "Switch to light theme" : "Switch to dark theme"); } if (btn) { btn.addEventListener("click", function () { var next = current() === "dark" ? "light" : "dark"; apply(next); try { localStorage.setItem(KEY, next); } catch (e) {} }); } // Follow OS changes only while the user hasn't made an explicit choice. try { var mq = window.matchMedia("(prefers-color-scheme: dark)"); mq.addEventListener("change", function (e) { var saved = null; try { saved = localStorage.getItem(KEY); } catch (err) {} if (!saved) apply(e.matches ? "dark" : "light"); }); } catch (e) {} })();
